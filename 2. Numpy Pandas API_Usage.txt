i) 1-D NUMPY:

import numpy as np
a = np.array([1,2,3,4])
type(a)		# numpy.ndarray
a.dtype		# dtype('int64')
a.size		# 4
a.ndim		# 1
a.shape		# (4,)

INDEXING AND SLICING:	a[3] = 99, b = a[0:3]

VECTOR ADDITION AND SUBTRACTION: u=[10,11], v=[21,30]  z = u + v   # z = [31, 41]
ARRAY ADDITION WITH SCALAR: x = np.array([1,2,3,4])    z = x + 5   # z = [6,7,8,9]
ARRAY MULTIPLICATION WITH SCALAR: y = np.array([1,2])  z = 2 * y   # z = [2,4]
MULTIPLICATION OF 2 ARRAYS:  x = [2,4] y = [3,6]       z = x * y   # z = [6, 24] 
DOT PRODUCT OF 2 ARRAYS: x = [2,4] y = [3,6] 	       z = np.dot(x, y)		# z = 6 + 24 = 30
